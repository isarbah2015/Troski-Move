3.2 Database Schema - Complete
Users Table
sql
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    email VARCHAR(255) UNIQUE,
    phone VARCHAR(20) UNIQUE NOT NULL,
    password_hash VARCHAR(255),
    first_name VARCHAR(100),
    last_name VARCHAR(100),
    national_id VARCHAR(20),
    profile_picture_url TEXT,
    role VARCHAR(50) DEFAULT 'commuter', -- commuter, driver, operator, admin, regulator
    is_verified BOOLEAN DEFAULT FALSE,
    is_active BOOLEAN DEFAULT TRUE,
    preferences JSONB,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    last_login TIMESTAMP,
    device_tokens JSONB,
    location GEOMETRY(Point, 4326),
    preferred_language VARCHAR(10) DEFAULT 'en',
    CONSTRAINT valid_email CHECK (email IS NULL OR email ~* '^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}$')
);

CREATE INDEX idx_users_phone ON users(phone);
CREATE INDEX idx_users_role ON users(role);
CREATE INDEX idx_users_location ON users USING GIST(location);
CREATE INDEX idx_users_created_at ON users(created_at DESC);
Vehicles Table
sql
CREATE TABLE vehicles (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    registration_number VARCHAR(20) UNIQUE NOT NULL,
    owner_id UUID REFERENCES users(id) ON DELETE CASCADE,
    vehicle_type VARCHAR(50) NOT NULL, -- trotro, bus, taxi, okada
    make VARCHAR(100),
    model VARCHAR(100),
    year INTEGER,
    color VARCHAR(50),
    capacity INTEGER NOT NULL,
    license_plate VARCHAR(20),
    insurance_policy VARCHAR(100),
    insurance_expiry DATE,
    roadworthy_certificate VARCHAR(100),
    roadworthy_expiry DATE,
    is_approved BOOLEAN DEFAULT FALSE,
    is_active BOOLEAN DEFAULT TRUE,
    last_inspection_date DATE,
    next_inspection_date DATE,
    inspection_score DECIMAL(5,2),
    gps_device_id VARCHAR(100),
    fuel_type VARCHAR(20) DEFAULT 'petrol', -- petrol, diesel, electric, hybrid
    battery_capacity DECIMAL(10,2), -- for EVs
    charging_ports VARCHAR(20),
    current_location GEOMETRY(Point, 4326),
    last_updated TIMESTAMP,
    metadata JSONB,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_vehicles_owner ON vehicles(owner_id);
CREATE INDEX idx_vehicles_registration ON vehicles(registration_number);
CREATE INDEX idx_vehicles_type ON vehicles(vehicle_type);
CREATE INDEX idx_vehicles_location ON vehicles USING GIST(current_location);
CREATE INDEX idx_vehicles_active ON vehicles(is_active) WHERE is_active = TRUE;
Routes Table
sql
CREATE TABLE routes (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(255) NOT NULL,
    start_location GEOMETRY(Point, 4326) NOT NULL,
    end_location GEOMETRY(Point, 4326) NOT NULL,
    start_address TEXT,
    end_address TEXT,
    distance_km DECIMAL(10,2),
    estimated_duration_minutes INTEGER,
    base_fare DECIMAL(10,2),
    fare_per_km DECIMAL(10,2),
    route_type VARCHAR(50) DEFAULT 'bus', -- bus, trotro, taxi, okada, rail
    status VARCHAR(50) DEFAULT 'active', -- active, inactive, maintenance
    is_approved BOOLEAN DEFAULT FALSE,
    average_daily_passengers INTEGER DEFAULT 0,
    peak_hours JSONB,
    special_fares JSONB,
    route_geometry GEOGRAPHY(LineString, 4326),
    waypoints JSONB,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_routes_type ON routes(route_type);
CREATE INDEX idx_routes_status ON routes(status);
CREATE INDEX idx_routes_location ON routes USING GIST(start_location);
CREATE INDEX idx_routes_geometry ON routes USING GIST(route_geometry);
Trips Table
sql
CREATE TABLE trips (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    route_id UUID REFERENCES routes(id) ON DELETE CASCADE,
    vehicle_id UUID REFERENCES vehicles(id) ON DELETE CASCADE,
    driver_id UUID REFERENCES users(id) ON DELETE CASCADE,
    scheduled_departure TIMESTAMP,
    actual_departure TIMESTAMP,
    scheduled_arrival TIMESTAMP,
    actual_arrival TIMESTAMP,
    status VARCHAR(50) DEFAULT 'scheduled', -- scheduled, in_progress, completed, cancelled, delayed
    available_seats INTEGER,
    occupied_seats INTEGER DEFAULT 0,
    current_location GEOMETRY(Point, 4326),
    route_progress DECIMAL(5,2) DEFAULT 0,
    delay_minutes INTEGER DEFAULT 0,
    cancellation_reason TEXT,
    fare_adjustment DECIMAL(10,2) DEFAULT 0,
    total_revenue DECIMAL(10,2) DEFAULT 0,
    driver_earnings DECIMAL(10,2) DEFAULT 0,
    company_revenue DECIMAL(10,2) DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_trips_route ON trips(route_id);
CREATE INDEX idx_trips_vehicle ON trips(vehicle_id);
CREATE INDEX idx_trips_driver ON trips(driver_id);
CREATE INDEX idx_trips_status ON trips(status);
CREATE INDEX idx_trips_scheduled ON trips(scheduled_departure);
CREATE INDEX idx_trips_location ON trips USING GIST(current_location);
Bookings Table
sql
CREATE TABLE bookings (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    booking_type VARCHAR(50) DEFAULT 'single', -- single, group, corporate
    passenger_count INTEGER DEFAULT 1,
    total_fare DECIMAL(10,2) NOT NULL,
    payment_status VARCHAR(50) DEFAULT 'pending', -- pending, paid, failed, refunded
    booking_status VARCHAR(50) DEFAULT 'confirmed', -- confirmed, checked_in, cancelled, no_show
    transaction_id UUID,
    payment_method VARCHAR(50), -- mobile_money, card, cash, tap_and_go
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    checked_in_at TIMESTAMP,
    cancelled_at TIMESTAMP,
    cancellation_reason TEXT,
    special_requests TEXT,
    metadata JSONB
);

CREATE INDEX idx_bookings_trip ON bookings(trip_id);
CREATE INDEX idx_bookings_user ON bookings(user_id);
CREATE INDEX idx_bookings_status ON bookings(booking_status);
CREATE INDEX idx_bookings_payment ON bookings(payment_status);
CREATE INDEX idx_bookings_created ON bookings(created_at);

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

CREATE INDEX idx_payments_booking ON payments(booking_id);
